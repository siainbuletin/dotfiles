* Why this exists

- *Problem*: the report and lint buffers listed their keys in the header line.
  - The header line does not wrap, so a long list ran off the window.
  - The keys at the end of the list could not be read at all.
- *Fix*: show the keys in the echo area instead, the way =nano-help.el= of nano-emacs does.
  - The echo area has room for the whole list and gives it a line of its own.
  - It shows the keys of the buffer you are in and nothing else.
- Source of the idea: [[https://github.com/rougier/nano-emacs/blob/master/nano-help.el][nano-help.el]] (nano-emacs, Nicolas Rougier).

** What was taken and what was not

- Taken: the hints are a list of (key, label) pairs, with the key in bold and the label plain.
- Taken: the hints are shown by the buffer's mode, not typed into a header by each command.
- Not taken: nano-help's own mode-specific tables; each buffer here gives its own list.
- Changed: the hints appear after a short idle time, and only when the echo area is empty.
  - A message such as "Dropped 3 entries" is never overwritten by the hints.
  - The hints come back on their own once the message has gone.

* How to use it

- Turn on =sp-echo-hints-mode= in the buffer.
- Set =sp-echo-hints-function= (buffer-local) to a function that returns the hints.
  - A hint is a cons cell =(KEY . LABEL)=, both strings: =("RET" . "visit")=.
  - The function runs each time, so the list can depend on the state of the buffer.
- Or set =sp-echo-hints-list= (buffer-local) to a fixed list of hints.
- =sp-echo-hints-format= turns a list into the string; it is a plain function, so a command can show a hint list by itself with =sp-echo-hints-show=.

* Settings

- =sp-echo-hints-delay=: seconds of idle time before the hints are shown.
- =sp-echo-hints-separator=: the text between two hints.
- =sp-echo-hints-key= (face): the look of a key; bold, like nano-help.

#+begin_src emacs-lisp -i
(defgroup sp-echo-hints nil
  "Key hints in the echo area."
  :group 'convenience)

(defcustom sp-echo-hints-delay 0.4
  "Seconds of idle time before the key hints are shown."
  :type 'number
  :group 'sp-echo-hints)

(defcustom sp-echo-hints-separator "   "
  "Text between two hints."
  :type 'string
  :group 'sp-echo-hints)

(defface sp-echo-hints-key
  '((t :inherit (bold font-lock-keyword-face)))
  "Face of the key in a key hint."
  :group 'sp-echo-hints)

(defvar sp-echo-hints-mode)

(defvar-local sp-echo-hints-list nil
  "The hints of this buffer: a list of (KEY . LABEL) strings.")

(defvar-local sp-echo-hints-function nil
  "Function with no arguments returning the hints of this buffer.
It wins over `sp-echo-hints-list'.")
#+end_src

* Building the text

- =sp-echo-hints-format= keeps the hints that fit in the frame width.
  - Hints are dropped from the end, so the first ones (the ones used most) stay.
  - A "…" marks that some were left out.
- A hint whose key or label is nil is skipped, so a function can write =(and cond '("d" . "drop"))= in a list it builds.

#+begin_src emacs-lisp -i
(defun sp-echo-hints--one (hint)
  "The text of HINT, a cons cell (KEY . LABEL)."
  (concat (propertize (car hint) 'face 'sp-echo-hints-key) " " (cdr hint)))

(defun sp-echo-hints-format (hints &optional width)
  "Turn HINTS, a list of (KEY . LABEL), into one string no wider than WIDTH.
WIDTH is the frame width by default.  Hints that do not fit are left out and
the text ends with an ellipsis."
  (let* ((width (or width (frame-width)))
         (hints (cl-remove-if-not (lambda (h) (and (consp h) (car h) (cdr h))) hints))
         (shown nil) (used 0))
    (dolist (hint hints)
      (let ((cost (+ (string-width (car hint)) 1 (string-width (cdr hint))
                     (if shown (string-width sp-echo-hints-separator) 0))))
        (when (<= (+ used cost 2) width)
          (setq used (+ used cost))
          (push hint shown))))
    (setq shown (nreverse shown))
    (concat (mapconcat #'sp-echo-hints--one shown sp-echo-hints-separator)
            (and (< (length shown) (length hints)) " …"))))
#+end_src

* Showing the hints

- =sp-echo-hints-current= returns the hints of the current buffer: the function first, then the list.
- =sp-echo-hints-show= writes them to the echo area.
  - =message-log-max= is nil, so they do not fill =*Messages*=.
  - The text is given as an argument, so a "%" in a label is safe.

#+begin_src emacs-lisp -i
(defun sp-echo-hints-current ()
  "The hints of the current buffer."
  (if sp-echo-hints-function
      (funcall sp-echo-hints-function)
    sp-echo-hints-list))

(defun sp-echo-hints-show (&optional hints)
  "Show HINTS, or the hints of the current buffer, in the echo area."
  (interactive)
  (let ((text (sp-echo-hints-format (or hints (sp-echo-hints-current))))
        (message-log-max nil))
    (unless (string-empty-p text)
      (message "%s" text))))
#+end_src

* The mode

- =sp-echo-hints-mode= is a buffer-local minor mode with no keys.
- One idle timer serves every buffer; it runs once each time Emacs goes idle.
- The timer shows the hints only when:
  - the selected buffer has the mode on,
  - the minibuffer is not active,
  - the echo area has no message (so a command's own message is never replaced).
- Turning the mode off in the last buffer cancels the timer.
- An error in a hint function turns the mode off in that buffer and says so once, instead of failing on every idle.

#+begin_src emacs-lisp -i
(defvar sp-echo-hints--timer nil
  "The idle timer that shows the hints.")

(defun sp-echo-hints--idle ()
  "Show the hints of the selected buffer if it has any and the echo area is free."
  (when (and sp-echo-hints-mode
             (not (active-minibuffer-window))
             (not (current-message)))
    (condition-case err
        (sp-echo-hints-show)
      (error
       (sp-echo-hints-mode -1)
       (message "sp-echo-hints: turned off in this buffer: %s" (error-message-string err))))))

(defun sp-echo-hints--any-buffer-p ()
  "Non-nil when some buffer has `sp-echo-hints-mode' on."
  (cl-some (lambda (buf) (buffer-local-value 'sp-echo-hints-mode buf)) (buffer-list)))

(defun sp-echo-hints--stop-timer ()
  "Cancel the idle timer when no buffer needs it."
  (when (and sp-echo-hints--timer (not (sp-echo-hints--any-buffer-p)))
    (cancel-timer sp-echo-hints--timer)
    (setq sp-echo-hints--timer nil)))

;;;###autoload
(define-minor-mode sp-echo-hints-mode
  "Show the key hints of this buffer in the echo area when Emacs is idle."
  :lighter nil
  (if sp-echo-hints-mode
      (progn
        (unless sp-echo-hints--timer
          (setq sp-echo-hints--timer
                (run-with-idle-timer sp-echo-hints-delay t #'sp-echo-hints--idle)))
        (add-hook 'kill-buffer-hook #'sp-echo-hints--stop-timer-later nil t))
    (remove-hook 'kill-buffer-hook #'sp-echo-hints--stop-timer-later t)
    (sp-echo-hints--stop-timer)))

(defun sp-echo-hints--stop-timer-later ()
  "Called when a buffer with the mode is killed: stop the timer if it was the last."
  (let ((buf (current-buffer)))
    (run-at-time 0 nil
                 (lambda ()
                   (unless (cl-some (lambda (b) (and (not (eq b buf))
                                                     (buffer-local-value 'sp-echo-hints-mode b)))
                                    (buffer-list))
                     (when sp-echo-hints--timer
                       (cancel-timer sp-echo-hints--timer)
                       (setq sp-echo-hints--timer nil)))))))

(provide 'sp-echo-hints)
;;; sp-echo-hints.el ends here

#+begin_src emacs-lisp -i
;;; sp-echo-hints-test.el --- Tests for sp-echo-hints -*- lexical-binding: t; -*-

(require 'ert)
(require 'sp-echo-hints)

(ert-deftest sp-echo-hints-format-joins-hints ()
  (should (equal (substring-no-properties
                  (sp-echo-hints-format '(("RET" . "visit") ("q" . "quit")) 80))
                 "RET visit   q quit")))

(ert-deftest sp-echo-hints-format-bolds-keys ()
  (let ((text (sp-echo-hints-format '(("RET" . "visit")) 80)))
    (should (eq (get-text-property 0 'face text) 'sp-echo-hints-key))
    (should-not (get-text-property 4 'face text))))

(ert-deftest sp-echo-hints-format-skips-empty-hints ()
  (should (equal (substring-no-properties
                  (sp-echo-hints-format '(("a" . "apply") nil ("d" . nil) ("q" . "quit")) 80))
                 "a apply   q quit")))

(ert-deftest sp-echo-hints-format-drops-what-does-not-fit ()
  (let ((text (substring-no-properties
               (sp-echo-hints-format '(("RET" . "visit") ("d" . "drop") ("q" . "quit")) 16))))
    (should (equal text "RET visit …"))
    (should (<= (string-width text) 16))))

(ert-deftest sp-echo-hints-current-prefers-function ()
  (with-temp-buffer
    (setq sp-echo-hints-list '(("a" . "list"))
          sp-echo-hints-function (lambda () '(("b" . "function"))))
    (should (equal (sp-echo-hints-current) '(("b" . "function"))))
    (setq sp-echo-hints-function nil)
    (should (equal (sp-echo-hints-current) '(("a" . "list"))))))

(ert-deftest sp-echo-hints-show-does-not-log-or-format ()
  "In batch mode `current-message' is always nil, so `message' is stubbed."
  (let (seen log-max)
    (cl-letf (((symbol-function 'message)
               (lambda (fmt &rest args)
                 (setq log-max message-log-max
                       seen (apply #'format-message fmt args)))))
      (let ((message-log-max 100))
        (sp-echo-hints-show '(("%" . "100%")))))
    (should (equal (substring-no-properties seen) "% 100%"))
    (should-not log-max)))

(ert-deftest sp-echo-hints-mode-starts-and-stops-the-timer ()
  (let ((sp-echo-hints--timer nil))
    (with-temp-buffer
      (sp-echo-hints-mode 1)
      (should (timerp sp-echo-hints--timer))
      (sp-echo-hints-mode -1)
      (should-not sp-echo-hints--timer))))

(ert-deftest sp-echo-hints-idle-leaves-a-message-alone ()
  (let (shown (current "Something else"))
    (cl-letf (((symbol-function 'current-message) (lambda () current))
              ((symbol-function 'message)
               (lambda (fmt &rest args) (setq shown (apply #'format-message fmt args)))))
      (with-temp-buffer
        (setq sp-echo-hints-list '(("a" . "apply")))
        (sp-echo-hints-mode 1)
        (sp-echo-hints--idle)
        (should-not shown)
        (setq current nil)
        (sp-echo-hints--idle)
        (should (equal (substring-no-properties shown) "a apply"))
        (sp-echo-hints-mode -1)))))

(provide 'sp-echo-hints-test)
