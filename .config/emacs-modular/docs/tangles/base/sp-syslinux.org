Literate source for =libraries/modules/base/sp-syslinux.el=.

- Every =emacs-lisp= block below goes into that file, except blocks marked =:tangle no=.
- The =:tangle= path in the =#+PROPERTY= line at the top says where they are written.
- To tangle, see [[file:../../../README.org::*Tangling the literate files][Tangling the literate files]] in the README.

This module is the Linux counterpart of =sp-syswin=, which handles running Emacs on Windows.

- It holds what only applies when Emacs runs on Linux (=system-type= is =gnu/linux=).
- For now that is the light or dark appearance of the desktop, which =sp-theme-auto-mode= (see [[file:../ui/sp-themes.org][sp-themes.org]]) follows. The mode asks for it through =sp-theme-system-providers= and =sp-theme-system-watchers=, and this module fills in the Linux entries, so Linux and Windows can later be used interchangeably with macOS.

Status:

- Loaded from =init.el=, next to =sp-macos=. Everything it does is guarded by =system-type=, so it is harmless elsewhere.

* Configuration

** The desktop appearance

The appearance is read from the first of these that answers:

- the freedesktop portal (=org.freedesktop.appearance color-scheme=, asked with =gdbus=), which GNOME 42 and later, KDE Plasma 5.24 and later and others implement: 1 is dark, 2 is light, 0 means no preference (no answer);
- =gsettings get org.gnome.desktop.interface color-scheme=: =prefer-dark=, =prefer-light= or =default= (light, as in GNOME);
- the KDE colour scheme name (=kreadconfig6= or =kreadconfig5=), dark when the name contains Dark.

When none of them answers (a window manager without a settings daemon, no D-Bus), the result is nil and the theme follows the time of day.

The watcher listens for the portal's =SettingChanged= signal through Emacs' own =dbus= library, so a change in the desktop settings reaches Emacs at once and no timer is needed. The parsing is kept in small functions so it can be tested without a desktop.

#+begin_src emacs-lisp -i
;;;; Desktop appearance
(defun sp-syslinux--parse-portal (output)
  "Return `light', `dark' or nil from the gdbus OUTPUT of the portal."
  (when (string-match "uint32 \\([0-9]+\\)" output)
    (pcase (match-string 1 output)
      ("1" 'dark)
      ("2" 'light))))

(defun sp-syslinux--parse-gsettings (output)
  "Return `light', `dark' or nil from the gsettings color-scheme OUTPUT."
  (cond ((string-match-p "prefer-dark" output) 'dark)
        ((string-match-p "prefer-light\\|default" output) 'light)))

(defun sp-syslinux--parse-kde (output)
  "Return `light', `dark' or nil from the name of a KDE colour scheme."
  (let ((name (string-trim output)))
    (unless (string-empty-p name)
      (if (string-match-p "dark" (downcase name)) 'dark 'light))))

(defun sp-syslinux--run (program &rest args)
  "Return the output of PROGRAM with ARGS, or nil if it fails or is missing."
  (when (executable-find program)
    (with-temp-buffer
      (when (zerop (apply #'call-process program nil (list (current-buffer) nil) nil args))
        (buffer-string)))))

(defun sp-syslinux-appearance ()
  "Return the desktop appearance, `light' or `dark', or nil if unknown."
  (when (eq system-type 'gnu/linux)
    (or (let ((out (sp-syslinux--run
                    "gdbus" "call" "--session"
                    "--dest" "org.freedesktop.portal.Desktop"
                    "--object-path" "/org/freedesktop/portal/desktop"
                    "--method" "org.freedesktop.portal.Settings.Read"
                    "org.freedesktop.appearance" "color-scheme")))
          (and out (sp-syslinux--parse-portal out)))
        (let ((out (sp-syslinux--run "gsettings" "get" "org.gnome.desktop.interface"
                                     "color-scheme")))
          (and out (sp-syslinux--parse-gsettings out)))
        (cl-some (lambda (program)
                   (let ((out (sp-syslinux--run program "--group" "General"
                                                "--key" "ColorScheme")))
                     (and out (sp-syslinux--parse-kde out))))
                 '("kreadconfig6" "kreadconfig5")))))

(defun sp-syslinux--signal-appearance (value)
  "Return `light', `dark' or nil for VALUE, the portal's color-scheme."
  (pcase (if (consp value) (car value) value)
    (1 'dark)
    (2 'light)))

(defun sp-syslinux-watch-appearance (callback)
  "Call CALLBACK with `light' or `dark' when the desktop flips.
Return the function that stops watching, or nil without D-Bus."
  (when (and (eq system-type 'gnu/linux)
             (require 'dbus nil t)
             (fboundp 'dbus-register-signal)
             (getenv "DBUS_SESSION_BUS_ADDRESS"))
    (let ((object
           (ignore-errors
             (dbus-register-signal
              :session "org.freedesktop.portal.Desktop"
              "/org/freedesktop/portal/desktop"
              "org.freedesktop.portal.Settings" "SettingChanged"
              (lambda (namespace key value)
                (when (and (equal namespace "org.freedesktop.appearance")
                           (equal key "color-scheme"))
                  (let ((appearance (sp-syslinux--signal-appearance value)))
                    (when appearance
                      (funcall callback appearance)))))))))
      (when object
        (lambda () (dbus-unregister-object object))))))

;; Follow the desktop appearance on Linux.
(when (and (eq system-type 'gnu/linux) (fboundp 'sp-theme-auto-mode))
  (sp-theme-auto-mode 1))

* Tests
:PROPERTIES:
:header-args:emacs-lisp: :tangle ../../../tests/sp-syslinux-test.el :comments no :padline yes :noweb no :mkdirp yes
:END:

They tangle to =tests/sp-syslinux-test.el= and run headless with =emacs -Q --batch -l tests/sp-syslinux-test.el -f ert-run-tests-batch-and-exit= from the folder of =init.el=. They test the parsing of what the desktop prints, not the desktop.

#+begin_src emacs-lisp -i
;;; sp-syslinux-test.el --- Tests for sp-syslinux  -*- lexical-binding: t; -*-

;;; Code:

(require 'ert)
(load (expand-file-name
       "../libraries/modules/base/sp-syslinux.el"
       (file-name-directory (or load-file-name buffer-file-name)))
      nil t)

(ert-deftest sp-syslinux-test-portal ()
  (should (eq (sp-syslinux--parse-portal "(<<uint32 1>>,)") 'dark))
  (should (eq (sp-syslinux--parse-portal "(<<uint32 2>>,)") 'light))
  (should-not (sp-syslinux--parse-portal "(<<uint32 0>>,)"))
  (should-not (sp-syslinux--parse-portal "")))

(ert-deftest sp-syslinux-test-gsettings ()
  (should (eq (sp-syslinux--parse-gsettings "'prefer-dark'\n") 'dark))
  (should (eq (sp-syslinux--parse-gsettings "'prefer-light'\n") 'light))
  (should (eq (sp-syslinux--parse-gsettings "'default'\n") 'light))
  (should-not (sp-syslinux--parse-gsettings "")))

(ert-deftest sp-syslinux-test-kde ()
  (should (eq (sp-syslinux--parse-kde "BreezeDark\n") 'dark))
  (should (eq (sp-syslinux--parse-kde "BreezeLight\n") 'light))
  (should-not (sp-syslinux--parse-kde "\n")))

(ert-deftest sp-syslinux-test-signal ()
  (should (eq (sp-syslinux--signal-appearance 1) 'dark))
  (should (eq (sp-syslinux--signal-appearance '(2)) 'light))
  (should-not (sp-syslinux--signal-appearance 0)))

(ert-deftest sp-syslinux-test-missing-programs ()
  (should-not (sp-syslinux--run "no-such-program-here")))

(provide 'sp-syslinux-test)
;;; sp-syslinux-test.el ends here
#+end_src
