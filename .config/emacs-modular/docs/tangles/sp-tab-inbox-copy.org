Literate source for =libraries/modules/text/sp-tab-inbox-copy.el=. Every line of the module lives in the =emacs-lisp= blocks below, in order. Tangling this file (=C-c C-v t=) rewrites that =.el= file from these blocks; nothing is tangled unless you run it.

This file is an earlier snapshot of =sp-tab-inbox.el=. It defines a =vulpea-ui= sidebar widget that searches a folder of plain files (the "link inbox", typically saved browser tabs) for mentions of the current note's title and aliases, without adding those files to the Vulpea database. Each hit can be visited, or collected: moved under a heading of the current note with =sp-move-region-to-heading=, leaving a link behind.

It depends on =vulpea=, =vulpea-ui=, the =vui= component library, the author's =sp-move-region-to-heading= module, and the external =rg= (ripgrep) program. Note that this copy also ends with =(provide 'sp-tab-inbox)=, the same feature name as the live module. The current =sp-tab-inbox.el= has grown beyond it (a send-elsewhere command, automatic Resources headings, a See also backlink, an inbox-backlinks widget, and a default directory based on =org-directory=). Nothing loads this copy; =init.el= loads =sp-tab-inbox=. It is probably kept as a backup, and loading both would make the later one silently redefine the earlier.

* File header

** Header, package requirements and commentary

The header declares Emacs 29.1, Vulpea 2, =vulpea-ui= and =vui= as requirements. The commentary summarises the behaviour: hits are word-wrapped, matches that only occur inside an =id:= link back into the current note are skipped, =collect= moves a hit and keeps its list bullet, =collect all= asks for the target heading once, and inbox files opened just for the operation are saved and closed. It also suggests loading the file after =vulpea-ui= and pressing =g= in the sidebar to refresh.

;;; sp-tab-inbox.el --- Link inbox widget for vulpea-ui -*- lexical-binding: t; -*-

;; Package-Requires: ((emacs "29.1") (vulpea "2") (vulpea-ui "0") (vui "0"))

;;; Commentary:

;; A `vulpea-ui' sidebar widget that searches a directory of plain files
;; (the "link inbox") for mentions of the current note's title and aliases,
;; without adding those files to the vulpea database.
;;
;; - Each hit is shown on its own lines, with the body text word-wrapped.
;; - Hits whose only match sits inside an id: link into the current note's

;; - `collect' moves a hit under a heading of the current note using

** Bullet regexp

=sp-tab-inbox-bullet-re= matches a list bullet at the start of a line (=-=, =+=, numbered items, or an indented =*=) together with an optional checkbox. A star in column 0 is excluded because in Org that is a heading.

#+begin_src emacs-lisp -i
(defconst sp-tab-inbox-bullet-re
  "\\(?:[ \t]*[-+]\\|[ \t]*[0-9]+[.)]\\|[ \t]+\\*\\)[ \t]+\\(?:\\[[ xX-]\\][ \t]+\\)?"
  "Matches a list bullet (and optional checkbox) at the start of a line.
A column-0 `*' is an Org heading, so it is not treated as a bullet.")
#+end_src

* Searching

** Collecting the note's IDs

=sp-tab-inbox--note-ids= returns the note's own ID plus every =:ID:= property found in its file. If the file is open, it reads the buffer (widened) so unsaved edits count; otherwise it reads the file from disk.

#+begin_src emacs-lisp -i
;;;; Searching

(defun sp-tab-inbox--note-ids (note)
  "Return the ID of NOTE plus every :ID: property in NOTE's file."
  (let ((path (vulpea-note-path note))
        (ids (list (vulpea-note-id note))))
    (when (and path (file-readable-p path))
      (with-temp-buffer
        (if-let ((buf (find-buffer-visiting path)))
            (insert (with-current-buffer buf
                      (save-restriction
                        (widen)
                        (buffer-substring-no-properties (point-min) (point-max)))))
          (insert-file-contents path))
        (goto-char (point-min))
        (while (re-search-forward "^[ \t]*:ID:[ \t]+\\([^ \t\n]+\\)" nil t)
          (push (match-string 1) ids))))
    (delete-dups ids)))
#+end_src

** Ignoring already-linked mentions

=sp-tab-inbox--unlinked-p= removes every =id:= link pointing at one of those IDs from a line of text, then checks, case-insensitively, whether any search term still appears. This is how lines that already link to the note are filtered out.

#+begin_src emacs-lisp -i
(defun sp-tab-inbox--unlinked-p (text terms ids)
  "Non-nil if some term in TERMS occurs in TEXT outside links to any of IDS."
  (let* ((case-fold-search t)
         (link-re (concat "\\[\\[id:" (regexp-opt ids)
                          "\\]\\(?:\\[[^]]*\\]\\)?\\]"))
         (stripped (replace-regexp-in-string link-re "" text t t)))
    (seq-some (lambda (term) (string-match-p (regexp-quote term) stripped))
              terms)))
#+end_src

** Running ripgrep

=sp-tab-inbox-search= gathers the note's title and aliases, ignoring any shorter than three characters, and runs =rg= over the inbox directory as a case-insensitive, fixed-string, whole-word search. It parses each =file:line:text= result and keeps only unlinked mentions, returning a list of (file, line, text). If =rg= is not installed it quietly returns nothing.

#+begin_src emacs-lisp -i
(defun sp-tab-inbox-search (note)
  "Return list of (FILE LINE TEXT) mentioning NOTE's title or aliases.
Matches that only sit inside id: links into NOTE's file are dropped."
  (when-let* ((rg (executable-find "rg"))
              (terms (seq-filter
                      (lambda (s) (>= (length s) 3))
                      (cons (vulpea-note-title note)
                            (vulpea-note-aliases note)))))
    (with-temp-buffer
      (apply #'call-process rg nil t nil
             "--line-number" "--no-heading" "--with-filename"
             "--ignore-case" "--fixed-strings" "--word-regexp"
             (append (mapcan (lambda (term) (list "-e" term)) terms)
                     (list (expand-file-name sp-tab-inbox-dir))))
      (goto-char (point-min))
      (let ((ids (sp-tab-inbox--note-ids note))
            result)
        (while (re-search-forward "^\\(.+?\\):\\([0-9]+\\):\\(.*\\)$" nil t)
          (let ((text (string-trim (match-string 3))))
            (when (sp-tab-inbox--unlinked-p text terms ids)
              (push (list (match-string 1)
                          (string-to-number (match-string 2))
                          text)
                    result))))
        (nreverse result)))))
#+end_src

* Display helpers

** Wrapping hit text

=sp-tab-inbox-wrap= fills a hit's text to the configured width and indents each line by two spaces for display.

#+begin_src emacs-lisp -i
;;;; Display helpers

(defun sp-tab-inbox-wrap (text)
  "Fill TEXT to `sp-tab-inbox-fill-column' and indent it by two spaces."
  (with-temp-buffer
    (insert text)
    (let ((fill-column sp-tab-inbox-fill-column))
      (fill-region (point-min) (point-max)))
    (goto-char (point-min))
    (while (not (eobp))
      (insert "  ")
      (forward-line 1))
    (buffer-string)))
#+end_src

* Collecting

** Moving one line into the note

=sp-tab-inbox-collect= moves a single inbox line under a heading of the current note. The heading is asked for with =sp--read-org-heading= unless a target is passed in. Before moving, it checks the line still has the expected text and raises an error asking you to refresh if the file has changed. If the line starts with a bullet, the move starts at the bullet so it goes along with the text, and the same bullet is reinserted before the link left behind in the inbox. The inbox file is saved, and killed afterwards only if this command opened it. It returns the target so a caller can reuse it.

#+begin_src emacs-lisp -i
;;;; Collecting

(defun sp-tab-inbox-collect (file line text note &optional target)
  "Move LINE of FILE under a heading of NOTE with `sp-move-region-to-heading'.
The list bullet, if any, is moved with the text and also kept in FILE in
front of the link left behind.
TEXT is the hit text, checked against the file to catch stale line numbers.
TARGET is an optional (MARKER . PATH) to skip the heading prompt.
Return the target used."
  (let* ((target (or target (sp--read-org-heading (vulpea-note-path note))))
         (existing (find-buffer-visiting file))
         (buf (or existing (find-file-noselect file))))
    (unwind-protect
        (with-current-buffer buf
          (save-excursion
            (goto-char (point-min))
            (forward-line (1- line))
            (let ((bol (line-beginning-position))
                  (end (line-end-position)))
              (unless (equal (string-trim (buffer-substring-no-properties bol end))
                             text)
                (user-error "Line %d of %s changed; refresh the sidebar"
                            line (file-name-nondirectory file)))
              (goto-char bol)
              (let* ((bullet-end (and (looking-at sp-tab-inbox-bullet-re)
                                      (match-end 0)))
                     ;; with a bullet, start at the bullet (skip indentation)
                     (beg (if bullet-end
                              (progn (skip-chars-forward " \t") (point))
                            bol))
                     (bullet (and bullet-end
                                  (buffer-substring-no-properties beg bullet-end))))
                (goto-char beg)
                (sp-move-region-to-heading beg end (car target) (cdr target))
                ;; the link now sits at BEG; put the bullet back before it
                (when bullet
                  (goto-char beg)
                  (insert bullet)))))
          (save-buffer))
      ;; only kill buffers this command opened
      (when (and (not existing) (buffer-live-p buf))
        (kill-buffer buf)))
    target))
#+end_src

** Collect one

=sp-tab-inbox-collect-one= collects a single hit, bumps the generation counter and refreshes the sidebar.

#+begin_src emacs-lisp -i
(defun sp-tab-inbox-collect-one (file line text note)
  "Collect a single hit (FILE, LINE, TEXT) into NOTE and refresh the sidebar."
  (sp-tab-inbox-collect file line text note)
  (cl-incf sp-tab-inbox-generation)
  (vulpea-ui-sidebar-refresh))
#+end_src

** Collect all

=sp-tab-inbox-collect-all= asks for a heading once and moves every hit under it. As the comment explains, each moved line is replaced by a one-line link, so the stored line numbers of later hits remain correct. It then reports how many lines were collected and refreshes.

#+begin_src emacs-lisp -i
(defun sp-tab-inbox-collect-all (hits note)
  "Collect every hit in HITS under one heading of NOTE."
  (when hits
    (let ((target (sp--read-org-heading (vulpea-note-path note))))
      ;; Each moved line is replaced by a one-line link, so line numbers
      ;; in the inbox files stay valid between hits.
      (dolist (hit hits)
        (pcase-let ((`(,file ,line ,text) hit))
          (sp-tab-inbox-collect file line text note target)))
      (cl-incf sp-tab-inbox-generation)
      (message "Collected %d link inbox lines" (length hits))
      (vulpea-ui-sidebar-refresh))))
#+end_src

* Widget

** The Link inbox component

=sp-tab-inbox-widget= is a =vui= component. It reads the current note from the sidebar, runs the search through =vui-use-memo= keyed on the note and the generation counter (so it is not repeated on every redraw), and renders a collapsible widget titled Link inbox with a hit count. With hits, it shows a =collect all= button followed by, for each hit, a =file:line= button that visits it, a =collect= button, and the wrapped text in the mention-context face.

#+begin_src emacs-lisp -i
;;;; Widget

(vui-defcomponent sp-tab-inbox-widget ()
  "Plain-text mentions of the current note in the link inbox."
  :render
  (let ((note (use-vulpea-ui-note)))
    (when note
      (let ((hits (vui-use-memo (note sp-tab-inbox-generation)
                    (sp-tab-inbox-search note))))
        (vui-component 'vulpea-ui-widget
          :title "Link inbox"
          :count (length hits)
          :children
          (lambda ()
            (if (null hits)
                (vui-text "No mentions")
              (apply #'vui-vstack
                     (vui-button "collect all"
                                 :on-click (lambda ()
                                             (sp-tab-inbox-collect-all hits note)))
                     (mapcar
                      (lambda (hit)
                        (pcase-let ((`(,file ,line ,text) hit))
                          (vui-vstack
                           (vui-hstack
                            (vui-button
                             (format "%s:%d" (file-name-nondirectory file) line)
                             :on-click (lambda () (sp-tab-inbox-visit file line)))
                            (vui-text " ")
                            (vui-button
                             "collect"
                             :on-click (lambda ()
                                         (sp-tab-inbox-collect-one
                                          file line text note))))
                           (vui-text
                            (propertize (sp-tab-inbox-wrap text)
                                        'face 'vulpea-ui-mention-context-face)))))
                      hits)))))))))
#+end_src

** Registering the widget

Adds the widget to the sidebar under the name =link-inbox= with order 360, which places it just after the built-in unlinked-mentions widget (350).

  :component 'sp-tab-inbox-widget
  :order 360) ; right after unlinked-mentions (350)
