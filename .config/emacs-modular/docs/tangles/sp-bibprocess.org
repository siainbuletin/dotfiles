Literate source for =libraries/modules/academic/sp-bibprocess.el=. Every line of the module lives in the =emacs-lisp= blocks below, in order. Tangling this file (=C-c C-v t=) rewrites that =.el= file from these blocks; nothing is tangled unless you run it.

The original file has no newline after its last line. Org always ends a tangled file with one, so tangling adds that final newline and changes nothing else.

This module manages the BibTeX bibliography that the academic modules build on. It defines where the =.bib= files live (=sp-bibtex-dir=, =~/bibliography/= by default), keeps a list of them that is shared with Org's citation system (=org-cite-global-bibliography=) and with =citar= (=citar-bibliography=), configures the built-in =bibtex= mode for BibLaTeX, and adds commands to look up new entries online with the =biblio= package and to audit attachments.

It is not listed directly in the module profiles; instead =sp-academic= loads it with =require=. The second half of the file repeats the variable, the register command and the lookup helper verbatim under a "BIBLIOGRAPHY" heading, apparently pasted in from another source; the repeats are harmless (the later =defvar= is a no-op and the function bodies are identical) but could be removed.

** Bibliography directory

=sp-bibtex-dir= is a user option pointing at the folder holding BibTeX files and their attachments, =~/bibliography/= under the home directory. The concatenation produces a doubled slash, which Emacs normalises when the path is expanded.

#+begin_src emacs-lisp -i
(defcustom sp-bibtex-dir
  (concat (file-name-as-directory (getenv "HOME")) "/bibliography/")
  "Location of BibTeX files and attachments."
  :group 'siapip
  :type 'directory)
#+end_src

** List of BibTeX files

=sp-bibtex-files= is filled at load time with every =.bib= file in the directory whose name starts with a letter or digit, or left =nil= if the directory does not exist. The regular expression is loose (the =|= inside the brackets and the unescaped dot are matched literally or as any character) but works for ordinary names.

#+begin_src emacs-lisp -i
(defvar sp-bibtex-files
  (when (file-exists-p sp-bibtex-dir)
    (directory-files sp-bibtex-dir t "^[A-Z|a-z|0-9].+.bib$"))
  "List of BibTeX files. Use `sp-bibtex-register' to configure.")
#+end_src

** Register BibTeX files

=sp-bibtex-register= rescans =sp-bibtex-dir= and sets =sp-bibtex-files=, =org-cite-global-bibliography= and =citar-bibliography= to the same list, then echoes the registered files. Run it after adding or removing a =.bib= file so citations and =citar= see the change.

#+begin_src emacs-lisp -i
;;;###autoload
(defun sp-bibtex-register ()
  "Register the contents of the `sp-bibtex-dir' with `sp-bibtex-files`.
Use when adding or removing a BibTeX file from or to `sp-bibtex-dir'."
  (interactive)
  (when (file-exists-p sp-bibtex-dir)
    (let ((bib-files (directory-files sp-bibtex-dir t
				      "^[A-Z|a-z|0-9].+.bib$")))
      (setq sp-bibtex-files bib-files
  	    org-cite-global-bibliography bib-files
	    citar-bibliography bib-files)))
  (message "Registered:\n%s" (mapconcat #'identity sp-bibtex-files "\n")))
#+end_src

** Combined biblio lookup

=sp--bibtex-combined-biblio-lookup= offers the backends known to =biblio= (such as CrossRef or arXiv) plus an extra "DOI" entry in one completion prompt. Choosing DOI asks for a DOI and inserts its BibTeX directly; any other backend starts an interactive =biblio-lookup= search. It relies on the internal =biblio--named-backends=.

#+begin_src emacs-lisp -i
(defun sp--bibtex-combined-biblio-lookup ()
  "Combines `biblio-lookup' and `biblio-doi-insert-bibtex'."
  (require 'biblio)
  (let* ((dbs (biblio--named-backends))
         (db-list (append dbs '(("DOI" . biblio-doi-backend))))
         (db-selected (biblio-completing-read-alist
                       "Backend:"
                       db-list)))
    (if (eq db-selected 'biblio-doi-backend)
        (let ((doi (read-string "DOI: ")))
          (biblio-doi-insert-bibtex doi))
      (biblio-lookup db-selected))))
#+end_src

* BibTeX mode settings

** Citation key format

These =bibtex-autokey-= settings shape the keys generated by =bibtex-clean-entry= and friends: the first author's name directly followed by a four-digit year, with no separators and no title words, giving keys like =smith2020=.

#+begin_src emacs-lisp -i
(setq bibtex-autokey-name-separator nil
      bibtex-autokey-name-year-separator nil
      bibtex-autokey-year-length 4
      bibtex-autokey-names 1
      bibtex-autokey-names-stretch 1
      bibtex-autokey-titlewords 0)
#+end_src

** bibtex

Configures the built-in =bibtex= mode: the BibLaTeX dialect, two extra optional fields offered when creating entries (=keywords= and =file= for attachments), and alignment of field values at the equals sign. After =bibtex= loads it registers the bibliography files, and =C-c w b r= runs =sp-bibtex-register= at any time.

#+begin_src emacs-lisp -i
(use-package bibtex
  :ensure nil
  :custom
  (bibtex-dialect 'biblatex)
  (bibtex-user-optional-fields
   '(("keywords" "Keywords to describe the entry" "")
     ("file"     "Relative or absolute path to attachments" "" )))
  (bibtex-align-at-equal-sign t)
  :config
  (sp-bibtex-register)
  :bind
  (("C-c w b r" . sp-bibtex-register))
)
#+end_src

** Insert a lookup into a bibliography

=sp-bibtex-biblio-lookup= picks the target =.bib= file: the only one if there is a single file, the current buffer if already in =bibtex-mode=, or a completion prompt otherwise. It opens that file, moves to the end, runs the combined lookup and saves. If no BibTeX files are registered it just reports that.

#+begin_src emacs-lisp -i
;;;###autoload
(defun sp-bibtex-biblio-lookup ()
  "Insert Biblio search results into current buffer or select BibTeX file."
  (interactive)
  (if-let* ((current-mode major-mode)
	   sp-bibtex-files
	   (bibfiles (length sp-bibtex-files))
	   (bibfile (cond ((eq bibfiles 1) (car sp-bibtex-files))
			  ((equal major-mode 'bibtex-mode)
			   (buffer-file-name))
			  (t (completing-read
			      "Select BibTeX file:" sp-bibtex-files)))))
      (progn (find-file bibfile)
	     (goto-char (point-max))
	     (sp--bibtex-combined-biblio-lookup)
	     (save-buffer))
    (message "No BibTeX file(s) defined.")))
#+end_src

* Auditing attachments

Two reports that compare the files stored in =sp-bibtex-dir= against the =file= fields in the BibTeX entries.

** Collect attachment paths from entries

=sp--bibtex-extract-attachments= re-registers the bibliography and scans each =.bib= file for =file = {...}= fields, splitting multiple attachments on semicolons and expanding each path relative to =sp-bibtex-dir=.

#+begin_src emacs-lisp -i
;; Search for missing BibTeX attachments and filenames
(defun sp--bibtex-extract-attachments ()
  "Extract attachment file names from BibTeX files in `sp-bibtex-dir'."
  (sp-bibtex-register)
  (let ((attachments '()))
    (dolist (bibtex-file sp-bibtex-files)
      (with-temp-buffer
        (insert-file-contents bibtex-file)
        (goto-char (point-min))
        (while (re-search-forward "file.*=.*{\\([^}]+\\)}" nil t)
          (let ((file-paths (split-string (match-string 1)
                                          "[[:space:]]*;[[:space:]]*")))
            (dolist (file-path file-paths)
              (push (expand-file-name (string-trim file-path)
                                      sp-bibtex-dir)
                    attachments))))))
    attachments))
#+end_src

** Collect files on disk

=sp--bibtex-extract-files= lists every file under =sp-bibtex-dir= recursively, leaving out =.bib= and =.csl= files.

#+begin_src emacs-lisp -i
(defun sp--bibtex-extract-files ()
  "List files recursively in `sp-bibtex-dir', excluding `.bib' and `.csl'."
  (seq-remove (lambda (file)
                (or (string-suffix-p ".bib" file)
                    (string-suffix-p ".csl" file)))
              (mapcar 'expand-file-name
                      (directory-files-recursively sp-bibtex-dir ""))))
#+end_src

** Files with no entry

=sp-bibtex-missing-files= reports files on disk that no BibTeX entry refers to, printing the count and each path to the =*Messages*= buffer.

#+begin_src emacs-lisp -i
;;;###autoload
(defun sp-bibtex-missing-files ()
  "List BibTeX attachments not listed in a BibTeX file entry."
  (interactive)
  (let* ((files (sp--bibtex-extract-files))
         (attachments (sp--bibtex-extract-attachments))
         (missing (cl-remove-if
                   (lambda (f) (member f attachments)) files)))
    (message "%s files not registered in bibliography" (length missing))
    (dolist (file missing)
      (message file))))
#+end_src

** Entries with no file

=sp-bibtex-missing-attachments= does the reverse: it lists attachment paths named in entries that do not exist on disk. The count message calls them "BibTeX files", although it is counting attachment paths. Both reports pass each path to =message= as a format string, so a file name containing a percent sign would be misprinted.

#+begin_src emacs-lisp -i
;;;###autoload
(defun sp-bibtex-missing-attachments ()
  "List BibTeX file entries with missing attachment(s)."
  (interactive)
  (let* ((files (sp--bibtex-extract-files))
         (attachments (sp--bibtex-extract-attachments))
         (missing (cl-remove-if
                   (lambda (f) (member f files)) attachments)))
    (message "%s BibTeX files without matching attachment." (length missing))
    (dolist (file missing)
      (message file))))
#+end_src
