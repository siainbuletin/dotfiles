Literate source for =libraries/modules/ui/sp-faces.el=.

- Every line of the module lives in the =emacs-lisp= blocks below, in order.
- The =:tangle= path in the =#+PROPERTY= line at the top says where they are written.
- To tangle, see [[file:../../tangling-instructions.org][Tangling instructions]].

This module gathers settings that change how text and the cursor look: outline heading faces in Lisp files, SVG tags in code, TODO and number highlighting, a TODO navigation transient, pulsing the current line after jumps, and highlighting of edited regions with Goggles. It also defines a small =what-face= helper for inspecting faces.

Packages:

- Optional packages (=outline-minor-faces=, =svg-tag-mode=, =highlight-numbers=, =hl-todo=, =goggles=) are configured unconditionally. The exception is =svg-tag-mode=, which needs SVG support.
- Built-ins used: =reveal=, =wid-edit=, =pulse= and =transient=.
- The =sp-todo-transient= defined here is bound under the leader map in =sp-keys=.

** What face

=what-face= reports the face at point in the echo area, which is handy when tweaking themes. It reads the face at =(point)= rather than its =pos= argument; since =pos= comes from =interactive "d"= these are the same when called interactively.

#+begin_src emacs-lisp -i
;;;; What Face?
;; https://stackoverflow.com/a/66287459/6277148
(defun what-face (pos)
  "State the face at point POS in the minibuffer."
  (interactive "d")
  (let ((face (or (get-char-property (point) 'read-face-name)
                  (get-char-property (point) 'face))))
    (if face (message "Face: %s" face) (message "No face at %d" pos))))
#+end_src

** Reveal mode

Turns on =global-reveal-mode=, which temporarily shows invisible text around point, such as folded Org headings, while you move through it. With =reveal-auto-hide= nil, revealed text stays visible after point leaves.

#+begin_src emacs-lisp -i
;;;; Reveal Mode
;; Toggle uncloaking of invisible text near point, including folded org headlines (Reveal mode).
(use-package reveal
  :ensure nil
  :defer 1
  :config
  (setq reveal-auto-hide nil)
  (global-reveal-mode))
#+end_src

** SVG tags in code

In programming buffers, =svg-tag-mode= draws matching text as rounded SVG labels when Emacs supports SVG. Any =:WORD:= pattern becomes a tag with the =success= face, and the markers =TODO:=, =FIXME:=, =HACK:=, =NOTE:= and =DONE:= get their own colours.

#+begin_src emacs-lisp -i
;;;; SVG Library (For Tags/Labels/etc.)
;;;;; SVG Tag Mode
(use-package svg-tag-mode
  :ensure t
  :when (image-type-available-p 'svg)
  :hook (prog-mode . svg-tag-mode)
  :config
  (setq svg-tag-tags
        '(;; Replaces any occurence of :XXX: with a dynamic SVG tag displaying XXX
          ("\\(:[A-Z]+:\\)" . ((lambda (tag)
                                 (svg-tag-make tag :face 'success :inverse t :beg 1 :end -1))))
          ;; other tags
          ("DONE:"  . ((lambda (tag) (svg-tag-make "DONE:"  :face 'fringe  :inverse t ))))
          ("FIXME:" . ((lambda (tag) (svg-tag-make "FIXME:" :face 'error   :inverse t))))
          ("HACK:"  . ((lambda (tag) (svg-tag-make "HACK:"  :face 'warning :inverse t))))
          ("NOTE:"  . ((lambda (tag) (svg-tag-make "NOTE:"  :face 'warning :inverse t))))
          ("TODO:"  . ((lambda (tag) (svg-tag-make "TODO:"  :face 'warning :inverse t)))))))
#+end_src

** SVG tags for Org headings

=M-x sp-org-heading-svg-tags-mode= draws the tags at the end of an Org heading as stock =svg-tag-mode= tags, in the current buffer only. It is *off* by default and nothing hooks it, so startup is not affected. Each tag name is a tag in the =org-tag= face and the colons around it are hidden. It does not use the tag groups, their colours or the emoji, and it does not touch =#+filetags:= lines or TODO keywords, because a match only counts inside the tags group of a heading (group 5 of =org-complex-heading-regexp=). It stays off, with a message, in a terminal frame or without SVG support. It uses the =svg-tag-mode= declared above, with its own buffer-local =svg-tag-tags=.

#+begin_src emacs-lisp -i
;;;;; SVG tags for Org headings
(defvar svg-tag-tags)
(declare-function svg-tag-make "svg-tag-mode")
(declare-function svg-tag-mode "svg-tag-mode")
(defvar org-complex-heading-regexp)

(defun sp-org-heading-svg-tags--in-tags-p (pos)
  "Non-nil when POS is inside the tags of an Org heading, not in a title or keyword."
  (save-match-data
    (save-excursion
      (goto-char pos)
      (forward-line 0)
      (and (eq (char-after) ?*)
           (looking-at org-complex-heading-regexp)
           (match-beginning 5)
           (<= (match-beginning 5) pos)
           (< pos (match-end 5))))))

(defun sp-org-heading-svg-tags--chip (tag)
  "Stock `svg-tag-make' chip of TAG (match 1) in an Org heading, else nil."
  (when (and (display-graphic-p)
             (sp-org-heading-svg-tags--in-tags-p (match-beginning 1)))
    (svg-tag-make tag :face 'org-tag :inverse t)))

(defun sp-org-heading-svg-tags--hide (_colon)
  "Display nothing for a colon (match 1) around the tags of an Org heading."
  (when (and (display-graphic-p)
             (sp-org-heading-svg-tags--in-tags-p (match-beginning 1)))
    ""))

(define-minor-mode sp-org-heading-svg-tags-mode
  "Draw the tags of Org headings in this buffer as stock SVG tags.
Off unless switched on; nothing is enabled at start.  Only the tags at
the end of a heading change: file-header keywords (#+filetags:) and TODO
keywords are left alone."
  :group 'spcustom
  (cond
   ((not sp-org-heading-svg-tags-mode)
    (when (bound-and-true-p svg-tag-mode)
      (svg-tag-mode -1)))
   ((not (and (display-graphic-p) (image-type-available-p 'svg)
              (require 'svg-tag-mode nil t)))
    (setq sp-org-heading-svg-tags-mode nil)
    (message "SVG tags need a graphic frame with SVG and svg-tag-mode"))
   (t
    (setq-local svg-tag-tags
                '(("\\(:\\)[[:alnum:]_@#%]+" sp-org-heading-svg-tags--hide)
                  ("\\([[:alnum:]_@#%]+\\):" sp-org-heading-svg-tags--chip)
                  (":[ \t]*$" sp-org-heading-svg-tags--hide)))
    (svg-tag-mode 1))))
#+end_src

** Highlight TODO keywords

Enables =hl-todo-mode= in programming and Markdown buffers so keywords like TODO and FIXME stand out. Enabling it in Org is commented out, presumably because Org already styles its own TODO keywords.

This form has =:ensure nil=: the package is installed (and its colours and global mode set) by =sp-org-taskman-extra=, and a second explicit declaration is logged by Elpaca as =Duplicate item ID queued: hl-todo=. The hooks are only added here, and the package loads on first use.

#+begin_src emacs-lisp -i
(use-package hl-todo
  ;; Installed by `sp-org-taskman-extra'; a second :ensure t is a duplicate order.
  :ensure nil
  :defer t
  :commands hl-todo-mode
  :init
  ;; (add-hook 'org-mode-hook #'hl-todo-mode)
  (add-hook 'prog-mode-hook #'hl-todo-mode)
  (add-hook 'markdown-mode-hook #'hl-todo-mode))
#+end_src

** TODO navigation transient

=sp-todo-transient=:

- =n= and =p= jump to the next and previous TODO keyword. =o= lists them with occur. All stay in the menu. =q= quits.
- On invocation it turns =hl-todo-mode= on and registers a one-shot =transient-exit-hook= function that turns it off again and removes itself. So highlighting is only active while navigating.
- Bound to =s t= under the leader key in =sp-keys=.

#+begin_src emacs-lisp -i
(transient-define-prefix sp-todo-transient ()
  "TODO navigation."
  ["TODO"
   ("n" "Next"     hl-todo-next     :transient t)
   ("p" "Previous" hl-todo-previous :transient t)
   ("o" "Occur"    hl-todo-occur    :transient t)
   ("q" "Quit"     transient-quit-one)]
  (interactive)
  (hl-todo-mode 1)
  (let (cleanup)
    (setq cleanup
          (lambda ()
            (hl-todo-mode -1)
            (remove-hook 'transient-exit-hook cleanup)))
    (add-hook 'transient-exit-hook cleanup))
  (transient-setup 'sp-todo-transient))
#+end_src

** Pulse the current line

Uses the built-in =pulse= library, following Karthink's "batteries included" article.

- =C-<return>= runs =pulse-line=, which briefly highlights the current line.
- Once pulse is loaded, the line also pulses on any change of selected window, through =window-selection-change-functions=.
- That hook is added in =:config=, so it only takes effect after =pulse= has been loaded, for example by the first =C-<return>=.

This block used to advise =scroll-up-command=, =scroll-down-command=, =recenter-top-bottom= and =other-window= with =pulse-line= as well. That advice was removed.

- =pulse-line= is interactive, so its empty =(interactive)= spec replaced the advised command's own spec.
- So =C-x o= called =other-window= with no COUNT and failed with "Wrong number of arguments".
- Those commands already pulse through =pulsar= in =sp-windows=.

#+begin_src emacs-lisp -i
;;;;; Highlight Cursor Line with Pulse
;; From https://karthinks.com/software/batteries-included-with-emacs/
;; Replace external package with internal command

(use-package pulse
  :bind
  ("C-<return>" . pulse-line)
  :commands (pulse-line pulse-momentary-highlight-one-line)
  :config
  (setq pulse-delay 0.08)
  (defun pulse-line (&rest _)
    "Pulse the current line."
    (interactive)
    (pulse-momentary-highlight-one-line (point)))
  ;; pulse on window change
  (push 'pulse-line window-selection-change-functions))
#+end_src

* Mode line and header line faces

** Title in the header line

=sp-modeline-title= is the face of the title of an Org buffer in the header line (=sp-nano-style=). Only the text of the =#+title:= keyword is taken; the heading face of the theme (=org-document-title=) is not used. The face is bold and nothing else, so it fits the header; change it here or with =M-x customize-face RET sp-modeline-title=. The module =sp-nano-style= defines the same face as a fallback for when this module is not loaded; this definition replaces it.

#+begin_src emacs-lisp -i
(defface sp-modeline-title
  '((t :inherit bold))
  "Title of an Org buffer in the header line (the text of its `#+title:').
Used by `sp-nano-style'.  Only the text of the title is shown, not its face.")
#+end_src
