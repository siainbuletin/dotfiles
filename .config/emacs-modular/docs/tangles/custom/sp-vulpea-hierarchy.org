* Parent/child views of vulpea notes

** Context

My notes follow a Zettelkasten pattern. One file is one idea. A sub-note records its parents with a =- parents :: [[id:…][Title]]= line in its zeroth section, and vulpea stores that as metadata. A note can have several parents: either one link after another on the same line, separated by commas, or one =- parents ::= line per parent. =vulpea-graph= doesn't draw this hierarchy (it analyses links for retrieval and shows lists), so this module reads the =parents= metadata and builds the tree itself.

A vault of this size is too large to draw whole. So every command first asks for a *starting note* and a *depth*, and shows only that part: the starting note, its children, their children, down to the depth.

** The commands

| Command                              | What it shows                                                                  |
|--------------------------------------+--------------------------------------------------------------------------------|
| =vulpea-hierarchy-graph=             | An interactive graph in the browser: pan, zoom, click a box to fold it.         |
| =vulpea-hierarchy-outline-native=    | An outline of Emacs buttons. =TAB= or =[+]=/=[-]= expands, =RET= visits a note. |
| =vulpea-hierarchy-outline-vui=       | The same outline built with =vui=. A trial, to compare with the native one.     |
| =vulpea-hierarchy-svg=               | A Graphviz SVG file named =<date and time>-<note>-<depth>.svg=.                 |

Notes on the last level show how many children they hide (=+N= in the browser, a count in the outline). The browser graph can't load more, so run it again with a bigger depth; the outlines expand any note on demand.

*** Reading every parent of a note (lines 76–92)

This is where a note with several parents is handled. vulpea's own rule for several values is one =- parents ::= line per value, and the metadata then holds one entry per line. A single line that lists several links separated by commas stays one entry, and =vulpea-note-meta-get-list= with the =link= type only returns the first link of an entry. So this function reads every link itself.

- Lines 76-79 define the function: it takes a note and returns the ids of all its parents.
- Line 80 starts an empty list, =ids=.
- Lines 81-82 go through every value stored under the parent key. A repeated key gives several values, so repeated lines are covered here.
- Line 83 sets =pos=, where to search next, and =found=, which records whether this value had any link.
- Line 84 finds the next Org link in the value, starting at =pos=.
- Line 85 records that a link was found.
- Lines 86-88 take the link target and, when it starts with =id:=, add the id without the prefix.
- Line 89 moves =pos= past this link, so the loop goes on to the next link in the same value. That is what makes a comma-separated line give every parent.
- Lines 90-91 handle a value with no link at all: if it isn't blank, the trimmed text is taken as a bare id.
- Line 92 puts the ids back in document order, drops empty ones (such as =[[id:][Title]]=), and removes duplicates.

#+begin_src emacs-lisp
(defun vulpea-hierarchy--parent-ids (note)
  "Ids of every parent NOTE names under `vulpea-hierarchy-parent-key'.
A value may hold several links, separated by commas or not, and the key
may repeat.  Every link counts, not only the first one in a line."
  (let (ids)
    (dolist (value (cdr (assoc vulpea-hierarchy-parent-key
                               (vulpea-note-meta note))))
      (let ((pos 0) (found nil))
        (while (string-match org-link-bracket-re value pos)
          (setq found t)
          (let ((link (match-string 1 value)))
            (when (string-prefix-p "id:" link)
              (push (string-remove-prefix "id:" link) ids)))
          (setq pos (match-end 0)))
        (when (and (not found) (not (string-empty-p (string-trim value))))
          (push (string-trim value) ids))))
    (delete-dups (seq-remove #'string-empty-p (nreverse ids)))))
#+end_src

#+begin_src emacs-lisp
(defun vulpea-hierarchy--build ()
  "Return a plist describing the whole hierarchy.
:notes maps id to note, :children maps parent id to child ids,
:roots lists ids that are parents but have no parent themselves."
  (let ((notes (make-hash-table :test #'equal))
        (children (make-hash-table :test #'equal))
        (has-parent (make-hash-table :test #'equal))
        parent-ids)
    (dolist (note (vulpea-db-query-by-meta-key vulpea-hierarchy-parent-key))
      (puthash (vulpea-note-id note) note notes)
      (dolist (pid (vulpea-hierarchy--parent-ids note))
        (puthash (vulpea-note-id note) t has-parent)
        (push (vulpea-note-id note) (gethash pid children))
        (push pid parent-ids)))
    ;; Parents that carry no parent key themselves are not loaded yet.
    (dolist (note (vulpea-db-query-by-ids
                   (seq-remove (lambda (id) (gethash id notes))
                               (delete-dups parent-ids))))
      (puthash (vulpea-note-id note) note notes))
    (list :notes notes
          :children children
          :roots (sort (seq-remove (lambda (id) (gethash id has-parent))
                                   (delete-dups (copy-sequence parent-ids)))
                       (lambda (a b)
                         (string< (vulpea-hierarchy--title a notes)
                                  (vulpea-hierarchy--title b notes)))))))

#+begin_src emacs-lisp
(defun vulpea-hierarchy--scope (tree start depth)
  "Return the part of TREE reached from START within DEPTH levels.
The result has the same keys as the tree, plus :more, which maps an
id at the last level to how many children it hides, and :nodes, the
ids in scope.  :roots is just (START)."
  (let ((notes (plist-get tree :notes))
        (children (make-hash-table :test #'equal))
        (more (make-hash-table :test #'equal))
        (seen (make-hash-table :test #'equal))
        (nodes (list start))
        (queue (list (cons start 0))))
    (puthash start t seen)
    (while queue
      (pcase-let ((`(,id . ,level) (pop queue)))
        (let ((kids (vulpea-hierarchy--sorted-children id tree)))
          (if (< level depth)
              (dolist (kid kids)
                (puthash id (append (gethash id children) (list kid)) children)
                (unless (gethash kid seen)
                  (puthash kid t seen)
                  (push kid nodes)
                  (setq queue (append queue (list (cons kid (1+ level)))))))
            (when kids
              (puthash id (length kids) more))))))
    (list :notes notes
          :children children
          :more more
          :roots (list start)
          :nodes (nreverse nodes))))

(defun vulpea-hierarchy--html (scope start depth)
  "Return a self-contained HTML page showing SCOPE from START to DEPTH."
  (let* ((notes (plist-get scope :notes))
         (nodes
          (vconcat
           (mapcar
            (lambda (id)
              (let ((note (vulpea-hierarchy--note id notes)))
                `((id . ,id)
                  (title . ,(vulpea-hierarchy--title id notes))
                  (path . ,(and note (vulpea-note-path note)))
                  (tags . ,(vconcat (and note (vulpea-note-tags note))))
                  (more . ,(gethash id (plist-get scope :more) 0)))))
            (plist-get scope :nodes))))
         (edges
          (vconcat
           (cl-loop for (parent . kids) in
                    (let (acc)
                      (maphash (lambda (k v) (push (cons k v) acc))
                               (plist-get scope :children))
                      acc)
                    append (mapcar (lambda (kid) (vector parent kid)) kids))))
         (data `((root . ,start)
                 (depth . ,depth)
                 (nodes . ,nodes)
                 (edges . ,edges))))
    (string-replace "__DATA__" (vulpea-hierarchy--json-for-html data)
                    (vulpea-hierarchy--html-template))))

(defun vulpea-hierarchy-view--expand-to-depth ()
  "Expand every node in the view above the starting depth."
  (let ((queue (list (cons vulpea-hierarchy--start 0))))
    (while queue
      (pcase-let ((`(,id . ,level) (pop queue)))
        (when (and (< level vulpea-hierarchy--depth)
                   (vulpea-hierarchy--sorted-children id vulpea-hierarchy--tree))
          (puthash id t vulpea-hierarchy--expanded)
          (dolist (kid (vulpea-hierarchy--sorted-children id vulpea-hierarchy--tree))

(defun vulpea-hierarchy-view--insert (id level path)
  "Insert the line for ID at indent LEVEL, then its children if expanded.
PATH holds the ancestors, so a cycle stops the recursion."
  (let* ((tree vulpea-hierarchy--tree)
         (notes (plist-get tree :notes))
         (kids (vulpea-hierarchy--sorted-children id tree))
         (open (and kids (gethash id vulpea-hierarchy--expanded)))
         (note (vulpea-hierarchy--note id notes)))
    (let ((start (point)))

*** Wrapping and labelling for Graphviz (lines 442–471)
Graphviz doesn't wrap text, so the module does it before writing the label. The browser page does its own wrapping in JavaScript, because only the browser knows how wide the text is.

- Line 442 is the section comment.
- =vulpea-hierarchy--wrap= cuts a text into lines of at most =width= characters. It fills a line word by word, starts a new line when the next word wouldn't fit, and cuts a word that is longer than a whole line.
- =vulpea-hierarchy--label= returns the title followed by the file tags of the note, written as =:tag1:tag2:=. A note without tags, or a missing one, gets just the title.

#+begin_src emacs-lisp
;;;; Static SVG export (Graphviz)

(defun vulpea-hierarchy--wrap (text width)
  "Return TEXT as a list of lines no longer than WIDTH, breaking long words."
  (let (lines (line ""))
    (dolist (word (split-string text))
      (while (> (length word) width)
        (when (> (length line) 0) (push line lines) (setq line ""))
        (push (substring word 0 width) lines)
        (setq word (substring word width)))
      (cond ((string-empty-p line) (setq line word))
            ((<= (+ (length line) 1 (length word)) width)
             (setq line (concat line " " word)))
            (t (push line lines) (setq line word))))
    (unless (string-empty-p line) (push line lines))
    (nreverse (or lines (list "")))))

(defun vulpea-hierarchy--label (id notes)
  "Return the title of ID from NOTES followed by its file tags, like :a:b:."
  (let* ((note (vulpea-hierarchy--note id notes))
         (tags (and note (vulpea-note-tags note))))
    (concat (vulpea-hierarchy--title id notes)
            (and tags (format " :%s:" (string-join tags ":"))))))

(defun vulpea-hierarchy--dot-label (id notes)
  "Return the wrapped, escaped Graphviz label text for ID."
  (mapconcat (lambda (line) (replace-regexp-in-string "[\"\\\\]" "\\\\\\&" line))
             (vulpea-hierarchy--wrap (vulpea-hierarchy--label id notes)
                                     vulpea-hierarchy-wrap-width)
             "\\n"))

function tokensOf(n) {
  const t = n.title.split(/\s+/).filter(Boolean).map(w => ({ text: w, cls: '' }));
  if (n.tags && n.tags.length) t.push({ text: ':' + n.tags.join(':') + ':', cls: 'tags' });
  if (n.more) t.push({ text: '+' + n.more, cls: 'more' });
  return t;
}

function breakToken(tok) {
  const out = [];
  let rest = tok.text;
  while (measure(rest) > TEXT_W && rest.length > 1) {
    let i = rest.length;
    while (i > 1 && measure(rest.slice(0, i)) > TEXT_W) i--;
    out.push({ text: rest.slice(0, i), cls: tok.cls });
    rest = rest.slice(i);
  }
  out.push({ text: rest, cls: tok.cls });
  return out;
}

*** Script: wrapping (lines 67–80)
Lines 67-80 define =wrap=. It places the pieces on lines one at a time, with a space between them (70-77). When the next piece would pass =TEXT_W=, it starts a new line. The result is a list of lines, each a list of pieces.

#+begin_src html :tangle ../../../libraries/modules/custom/sp-vulpea-hierarchy.html :comments no :padline yes :mkdirp yes
function wrap(tokens) {
  const lines = [], space = measure(' ');
  let line = [], width = 0;
  for (const tok0 of tokens) {
    for (const tok of breakToken(tok0)) {
      const w = measure(tok.text);
      if (line.length && width + space + w > TEXT_W) { lines.push(line); line = []; width = 0; }
      width += (line.length ? space : 0) + w;
      line.push(tok);
    }
  }
  if (line.length) lines.push(line);
  return lines;
}

#+begin_src html :tangle ../../../libraries/modules/custom/sp-vulpea-hierarchy.html :comments no :padline yes :mkdirp yes
function layout(id, depth, path, cursor) {
  const node = { id, depth, kids: [], y: 0, h: info(id).h };
  if (!path.includes(id)) {
    for (const c of childrenOf(id)) node.kids.push(layout(c, depth + 1, [...path, id], cursor));
  }
  if (node.kids.length) {
    const first = node.kids[0], last = node.kids[node.kids.length - 1];
    node.y = (first.y + first.h / 2 + last.y + last.h / 2) / 2 - node.h / 2;
    cursor.y = Math.max(cursor.y, node.y + node.h + GAP);
  } else {
    node.y = cursor.y;
    cursor.y += node.h + GAP;
  }
  return node;
}

function topOf(node) { return Math.min(node.y, ...node.kids.map(topOf)); }

  const { n, lines } = info(node.id);
  const x = node.depth * COL, y = node.y + shift;
  if (parent) {
    const px = parent.x + W, py = parent.y + parent.h / 2, cy = y + node.h / 2;
    el('path', { class: 'edge', d: `M${px},${py} C${px + 40},${py} ${x - 40},${cy} ${x},${cy}` }, vp);

  el('rect', { width: W, height: node.h, rx: 6 }, g);
  lines.forEach((line, i) => {
    const t = el('text', { x: PAD, y: VPAD + LINE * i + 13 }, g);
    line.forEach((tok, j) => {
      const s = el('tspan', tok.cls ? { class: tok.cls } : {}, t);
