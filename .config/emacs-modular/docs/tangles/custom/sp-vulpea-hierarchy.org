* Parent/child views of vulpea notes

** Context

My notes follow a Zettelkasten pattern. One file is one idea. A sub-note records its parents with a =- parents :: [[id:…][Title]]= line in its zeroth section, and vulpea stores that as metadata. A note can have several parents: either one link after another on the same line, separated by commas, or one =- parents ::= line per parent. =vulpea-graph= doesn't draw this hierarchy (it analyses links for retrieval and shows lists), so this module reads the =parents= metadata and builds the tree itself.

Besides =parents=, a note can relate to other notes in other ways: =supports=, =opposes=, =informs=, =answers=, or any relation you declare in =vulpea-hierarchy-relations=. Each relation lists the places it is read from: a metadata key on the note, the inverse of a key on the other note (=children= for =parents=), or a file property such as =DG_SUPPORTS=, which is how =discourse-graphs= stores its relations. Parents and children build the tree. The other relations are drawn as labelled, coloured, dashed links between two notes that are both in view.

A vault of this size is too large to draw whole. So every command first asks for a *starting note* and a *depth*, and shows only that part: the starting note, its children, their children, down to the depth.

** The commands

| Command                              | What it shows                                                                  |
|--------------------------------------+--------------------------------------------------------------------------------|
| =vulpea-hierarchy-graph=             | An interactive graph in the browser: pan, zoom, click a box to fold it.         |
| =vulpea-hierarchy-outline-native=    | An outline of Emacs buttons. =TAB= or =[+]=/=[-]= expands, =RET= visits a note. |
| =vulpea-hierarchy-outline-vui=       | The same outline built with =vui=. A trial, to compare with the native one.     |
| =vulpea-hierarchy-svg=               | A Graphviz SVG file named =<date and time>-<note>-<depth>.svg=.                 |

Notes on the last level show how many children they hide (=+N= in the browser, a count in the outline). The browser graph can't load more, so run it again with a bigger depth; the outlines expand any note on demand.

(defun vulpea-hierarchy--note (id notes)

#+begin_src emacs-lisp
(defun vulpea-hierarchy--value-ids (values)
  "Return the ids named in VALUES, a list of strings.
A value may hold several links, separated by commas or not.  Every link
counts, not only the first one in a value.  A value with no link is
read as bare ids separated by commas or spaces."
  (let (ids)
    (dolist (value values)
      (let ((pos 0) (found nil))
        (while (string-match org-link-bracket-re value pos)
          (setq found t)
          (let ((link (match-string 1 value)))
            (when (string-prefix-p "id:" link)
              (push (string-remove-prefix "id:" link) ids)))
          (setq pos (match-end 0)))
        (unless found
          (dolist (word (split-string value "[ ,]+" t))
            (push word ids)))))
    (delete-dups (seq-remove #'string-empty-p (nreverse ids)))))

(defun vulpea-hierarchy--source-values (note source)
  "Return the strings that SOURCE finds in NOTE."
  (pcase source
    ((or `(meta ,key) `(meta-inverse ,key))
     (cdr (assoc key (vulpea-note-meta note))))
    ((or `(property ,key) `(property-inverse ,key))
     (when-let ((value (cdr (assoc key (vulpea-note-properties note)))))
       (list value)))))

(defun vulpea-hierarchy--source-notes (source)
  "Return the notes that may hold SOURCE."
  (pcase source
    ((or `(meta ,key) `(meta-inverse ,key))
     (vulpea-db-query-by-meta-key key))
    ((or `(property ,key) `(property-inverse ,key))
     (vulpea-db-query
      (lambda (note) (assoc key (vulpea-note-properties note)))))))

(defun vulpea-hierarchy--edges ()
  "Return every edge as (SUBJECT RELATION OBJECT), and the notes read.
The second value is a hash table from id to note."
  (let ((notes (make-hash-table :test #'equal))
        (seen (make-hash-table :test #'equal))
        edges)
    (pcase-dolist (`(,relation . ,plist) vulpea-hierarchy-relations)
      (dolist (source (plist-get plist :sources))
        (dolist (note (vulpea-hierarchy--source-notes source))
          (puthash (vulpea-note-id note) note notes)
          (dolist (id (vulpea-hierarchy--value-ids
                       (vulpea-hierarchy--source-values note source)))
            (let* ((inverse (memq (car source) '(meta-inverse property-inverse)))
                   (edge (if inverse
                             (list id relation (vulpea-note-id note))
                           (list (vulpea-note-id note) relation id))))
              (unless (gethash edge seen)
                (puthash edge t seen)
                (push edge edges)))))))
    (cons (nreverse edges) notes)))

#+begin_src emacs-lisp
(defun vulpea-hierarchy--build ()
  "Return a plist describing the whole hierarchy.
:notes maps id to note, :children maps parent id to child ids,
:roots lists ids that are parents but have no parent themselves, and
:links lists the edges (SUBJECT RELATION OBJECT) of every other relation."
  (pcase-let* ((`(,edges . ,notes) (vulpea-hierarchy--edges))
               (children (make-hash-table :test #'equal))
               (has-parent (make-hash-table :test #'equal))
               (parent-ids nil)
               (links nil)
               (ids nil))
    (pcase-dolist (`(,subject ,relation ,object) edges)
      (push subject ids)
      (push object ids)
      (if (eq relation 'parent)
          (progn (puthash subject t has-parent)
                 (push subject (gethash object children))
                 (push object parent-ids))
        (push (list subject relation object) links)))
    ;; Notes that only appear as the object of an edge are not loaded yet.
    (dolist (note (vulpea-db-query-by-ids
                   (seq-remove (lambda (id) (gethash id notes))
                               (delete-dups ids))))
      (puthash (vulpea-note-id note) note notes))
    (list :notes notes
          :children children
          :links (nreverse links)
          :roots (sort (seq-remove (lambda (id) (gethash id has-parent))
                                   (delete-dups (copy-sequence parent-ids)))
                       (lambda (a b)
                         (string< (vulpea-hierarchy--title a notes)
                                  (vulpea-hierarchy--title b notes)))))))

#+begin_src emacs-lisp
(defun vulpea-hierarchy--scope (tree start depth)
  "Return the part of TREE reached from START within DEPTH levels.
The result has the same keys as the tree, plus :more, which maps an
id at the last level to how many children it hides, and :nodes, the
ids in scope.  :roots is just (START).  :links keeps the other
relations between two notes that are both in scope."
  (let ((notes (plist-get tree :notes))
        (children (make-hash-table :test #'equal))
        (more (make-hash-table :test #'equal))
        (seen (make-hash-table :test #'equal))
        (nodes (list start))
        (queue (list (cons start 0))))
    (puthash start t seen)
    (while queue
      (pcase-let ((`(,id . ,level) (pop queue)))
        (let ((kids (vulpea-hierarchy--sorted-children id tree)))
          (if (< level depth)
              (dolist (kid kids)
                (puthash id (append (gethash id children) (list kid)) children)
                (unless (gethash kid seen)
                  (puthash kid t seen)
                  (push kid nodes)
                  (setq queue (append queue (list (cons kid (1+ level)))))))
            (when kids
              (puthash id (length kids) more))))))
    (setq nodes (nreverse nodes))
    (list :notes notes
          :children children
          :more more
          :roots (list start)
          :nodes nodes
          :links (seq-filter (pcase-lambda (`(,subject ,_ ,object))
                               (and (gethash subject seen) (gethash object seen)))
                             (plist-get tree :links)))))

(defun vulpea-hierarchy--read-scope ()
  "Ask for a starting note and a depth.  Return (START-ID . DEPTH)."

*** Wrapping and labelling for Graphviz (lines 550–579)
Graphviz doesn't wrap text, so the module does it before writing the label. The browser page does its own wrapping in JavaScript, because only the browser knows how wide the text is.

;;;; Static SVG export (Graphviz)

(defun vulpea-hierarchy--wrap (text width)
  "Return TEXT as a list of lines no longer than WIDTH, breaking long words."
  (let (lines (line ""))
    (dolist (word (split-string text))
      (while (> (length word) width)
        (when (> (length line) 0) (push line lines) (setq line ""))
        (push (substring word 0 width) lines)
        (setq word (substring word width)))
      (cond ((string-empty-p line) (setq line word))
            ((<= (+ (length line) 1 (length word)) width)
             (setq line (concat line " " word)))
            (t (push line lines) (setq line word))))
    (unless (string-empty-p line) (push line lines))
    (nreverse (or lines (list "")))))

function tokensOf(n) {
  const t = n.title.split(/\s+/).filter(Boolean).map(w => ({ text: w, cls: '' }));
  if (n.tags && n.tags.length) t.push({ text: ':' + n.tags.join(':') + ':', cls: 'tags' });
  if (n.more) t.push({ text: '+' + n.more, cls: 'more' });
  return t;
}

function breakToken(tok) {
  const out = [];
  let rest = tok.text;
  while (measure(rest) > TEXT_W && rest.length > 1) {
    let i = rest.length;
    while (i > 1 && measure(rest.slice(0, i)) > TEXT_W) i--;
    out.push({ text: rest.slice(0, i), cls: tok.cls });
    rest = rest.slice(i);
  }
  out.push({ text: rest, cls: tok.cls });
  return out;
}

*** Script: wrapping (lines 74–87)
Lines 74-87 define =wrap=. It places the pieces on lines one at a time, with a space between them (77-84). When the next piece would pass =TEXT_W=, it starts a new line. The result is a list of lines, each a list of pieces.

#+begin_src html :tangle ../../../libraries/modules/custom/sp-vulpea-hierarchy.html :comments no :padline yes :mkdirp yes
function wrap(tokens) {
  const lines = [], space = measure(' ');
  let line = [], width = 0;
  for (const tok0 of tokens) {
    for (const tok of breakToken(tok0)) {
      const w = measure(tok.text);
      if (line.length && width + space + w > TEXT_W) { lines.push(line); line = []; width = 0; }
      width += (line.length ? space : 0) + w;
      line.push(tok);
    }
  }
  if (line.length) lines.push(line);
  return lines;
}

function layout(id, depth, path, cursor) {
  const node = { id, depth, kids: [], y: 0, h: info(id).h };
  if (!path.includes(id)) {
    for (const c of childrenOf(id)) node.kids.push(layout(c, depth + 1, [...path, id], cursor));
  }
  if (node.kids.length) {
    const first = node.kids[0], last = node.kids[node.kids.length - 1];
    node.y = (first.y + first.h / 2 + last.y + last.h / 2) / 2 - node.h / 2;
    cursor.y = Math.max(cursor.y, node.y + node.h + GAP);
  } else {
    node.y = cursor.y;
    cursor.y += node.h + GAP;
  }
  return node;
}
