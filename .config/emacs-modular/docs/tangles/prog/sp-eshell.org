- Every line of the module lives in the =emacs-lisp= blocks below, in order.
- The =:tangle= path in the =#+PROPERTY= line at the top says where they are written.
- To tangle, see [[file:../../tangling-instructions.org][Tangling instructions]].

The original file has no newline after its last line. Org always ends a tangled file with one, so tangling adds that final newline and changes nothing else.

Configuration for Eshell, Emacs's built-in shell written in Lisp. It:

- tunes the built-in Eshell sub-packages (history, globbing, completion, visual commands and so on),
- adds optional completion and help packages,
- defines a two-line prompt with time, shortened path, git repository and branch, and a running command counter,
- sets up a list of aliases,
- makes =ls= output clickable with icons,
- adds many helper commands: a project-aware Eshell toggle, =less=, =info=, =z= directory jumping, a TRAMP-aware =rcd=, and more.

The module is adapted from Lambda-Emacs's eshell setup and is loaded in the Programming group of =init.el=. It uses the Lambda theme faces =lambda-meek=, =lambda-mild= and =lambda-yellow=, with fallbacks if the theme is missing, and =nerd-icons= for listings. 

Known problems:

- Several paths are built from =lem-etc-dir=. In this repository that constant appears to be defined only in =claude-early.el=, which =init.el= does not seem to load. If it is unbound, the history, alias and directory-ring settings would signal an error when Eshell first loads.
- =sp-workspaces= calls =sp-eshell-home=, which does not exist. The function here is named =lem-eshell-home=.

* Setup

** Header and fallback faces

Turns on lexical binding, then at both compile and load time tries to load =lambda-themes= and defines fallback versions of the three faces the prompt uses if they do not exist yet: =lambda-meek= and =lambda-mild= inherit =shadow=, and =lambda-yellow= inherits =warning=. This keeps the prompt working with any theme.

#+begin_src emacs-lisp -i
;;; sp-eshell.el --- Eshell setup, prompt, aliases and helpers -*- lexical-binding: t; -*-

;;; Commentary:

;; Configuration for Eshell, Emacs's built-in shell written in Lisp.

;;; Code:

;; Define required faces FIRST, before any other code
(eval-and-compile
  ;; Try to load lambda themes
  (require 'lambda-themes nil t)
  
  ;; Ensure all required faces are defined
  (unless (facep 'lambda-meek)
    (defface lambda-meek
      '((t :inherit shadow))
      "Fallback meek face for eshell."
      :group 'faces))
  
  (unless (facep 'lambda-mild)
    (defface lambda-mild  
      '((t :inherit shadow))
      "Fallback mild face for eshell."
      :group 'faces))
  
  (unless (facep 'lambda-yellow)
    (defface lambda-yellow
      '((t :inherit warning))
      "Fallback yellow face for eshell."
      :group 'faces)))
#+end_src

(add-hook 'eshell-exit-hook (lambda () (setq eshell-prompt-number 0)))
(advice-add 'eshell-send-input :before
            (lambda (&rest args) (setq eshell-prompt-number (+ 1 eshell-prompt-number))))

** Git segment

=lem-eshell-config--git-prompt= returns a segment like =(repo branch)= when the directory is local, =git= is installed and a =.git= is found above it.

- The repository name is derived from the =origin= remote URL, so a repository without a remote shows an empty name.
- It shells out to git twice per prompt.

#+begin_src emacs-lisp -i
(defun lem-eshell-config--git-prompt (pwd)
  "Returns current git branch as a string, or the empty string if
PWD is not in a git repo (or the git command is not found)."
  (interactive)
  (when (and (not (file-remote-p pwd))
             (eshell-search-path "git")
             (locate-dominating-file pwd ".git"))
    (let* ((git-url (shell-command-to-string "git config --get remote.origin.url"))
           (git-repo (file-name-base (string-trim git-url)))
           (git-output (shell-command-to-string (concat "git rev-parse --abbrev-ref HEAD")))
           (git-branch (string-trim git-output))
           (git-icon  "\xe0a0")
           (git-icon2 (propertize "\xf020" 'face `(:family "octicons")))
           (git-sep (propertize "" 'face 'lambda-meek)))
      (concat (propertize " (" 'face 'lambda-mild)
              (propertize git-repo 'face `(:inherit lambda-meek :weight light))
              git-sep
              (propertize git-branch 'face `(:inherit lambda-meek :weight light))
              (propertize ") " 'face 'lambda-mild)))))
#+end_src

** Home as tilde

=lem--pwd-replace-home= replaces the home directory prefix with =~=.

#+begin_src emacs-lisp -i
(defun lem--pwd-replace-home (pwd)
  "Replace home in PWD with tilde (~) character."
  (interactive)
  (let* ((home (expand-file-name (getenv "HOME")))
         (home-len (length home)))
    (if (and
         (>= (length pwd) home-len)
         (equal home (substring pwd 0 home-len)))
        (concat "~" (substring pwd home-len))
      pwd)))
#+end_src

** Shortening the path

=lem--pwd-shorten-dirs= abbreviates every path component to its first letter except the last two, so =~/projects/emacs/config= becomes =~/p/emacs/config=.

#+begin_src emacs-lisp -i
(defun lem--pwd-shorten-dirs (pwd)
  "Shorten all directory names in PWD except the last two."
  (let ((p-lst (split-string pwd "/")))
    (if (> (length p-lst) 2)
        (concat
         (mapconcat (lambda (elm) (if (zerop (length elm)) ""
                               (substring elm 0 1)))
                    (butlast p-lst 2)
                    "/")
         "/"
         (mapconcat (lambda (elm) elm)
                    (last p-lst 2)
                    "/"))
      pwd)))  ;; Otherwise, we just return the PWD
#+end_src

** Splitting parent and name

=lem--split-directory-prompt= splits the shortened path into its parent part and final directory name, so the name can be drawn in bold.

#+begin_src emacs-lisp -i
(defun lem--split-directory-prompt (directory)
  (if (string-match-p ".*/.*" directory)
      (list (file-name-directory directory) (file-name-base directory))
    (list "" directory)))
#+end_src

** The prompt function

=lem-eshell-config--prompt-function= builds a three-line prompt drawn with box characters:

- Line 1: an OS icon (an Apple SF Symbol on macOS, a penguin on Linux), the time, and the shortened path with the directory name in bold.
- Line 2: the command counter.
- Line 3: the git segment and the prompt character in bold yellow.

A final space with the =default= face keeps typed input from inheriting the prompt's face.

#+begin_src emacs-lisp -i
(defun lem-eshell-config--prompt-function ()
  "Prettify eshell prompt."
  (let* ((os-char (cond ((string-equal system-type "darwin") "􀣺")
                        ((string-equal system-type "gnu/linux") "🐧")
                        ((string-equal system-type "cygwin") "🗔")
                        (t "?")))
         (pwd        (eshell/pwd))
         (directory (lem--split-directory-prompt
                     (lem--pwd-shorten-dirs
                      (lem--pwd-replace-home pwd))))
         (parent (car directory))
         (name   (cadr directory))
         (branch (lem-eshell-config--git-prompt pwd)))

    (concat (propertize "\n╭─ " 'face 'lambda-mild)
            (propertize (format "%s" os-char) 'face 'lambda-meek)
            (propertize " ─ "  'face 'lambda-mild)
            (propertize (format-time-string "%H:%M:%S" (current-time))  'face 'lambda-meek)
            (propertize " ─ "  'face 'lambda-mild)
            (propertize parent 'face 'lambda-meek)
            (propertize name 'face `(:inherit lambda-meek :weight bold))
            "\n"
            (propertize (concat "│" (number-to-string eshell-prompt-number))   'face 'lambda-mild)
            "\n"
            (propertize "╰─>>"  'face 'lambda-mild)
            (if branch
                branch
              " ")
            (propertize (lem-eshell-config--prompt-char) 'face `(:inherit lambda-yellow :weight ultra-bold))
            ;; needed for the input text to not have prompt face
            (propertize " " 'face 'default))))
#+end_src

** Default aliases

=lem-eshell-aliases= is the alias list:

- git shortcuts (=g=, =gs=, =gd=, =gco= and others), with =gg= and =gl= opening Magit,
- Homebrew shortcuts,
- =ls= variants that group by extension,
- navigation shortcuts that open Emacs commands: =d= for Dired, =ff= and =e= for =find-file=, =bb= for =consult-buffer=, =pp= to switch project, =up= and =bd= for =eshell-up=,

** Defining aliases from Lisp

=lem-set-eshell-alias= takes alternating alias names and commands.

- Updates or adds them in =lem-eshell-aliases=.
- Once =em-alias= is loaded, rebuilds =eshell-command-aliases-list= from the saved defaults plus this list.
- Errors on an odd number of arguments.
- Nothing in the config currently calls it.

#+begin_src emacs-lisp -i
;;;###autoload
(defun lem-set-eshell-alias (&rest aliases)
  "Define aliases for eshell.
ALIASES is a flat list of alias -> command pairs. e.g.
  (lem-set-eshell-alias
    \"hi\"  \"echo hello world\"
    \"bye\" \"echo goodbye world\")"
  (or (cl-evenp (length aliases))
      (signal 'wrong-number-of-arguments (list 'even (length aliases))))
  (with-eval-after-load 'em-alias
    (while aliases
      (let ((alias (pop aliases))
            (command (pop aliases)))
        (if-let* ((oldval (assoc alias lem-eshell-aliases)))
            (setcdr oldval (list command))
          (push (list alias command) lem-eshell-aliases))))
    (when (boundp 'eshell-command-aliases-list)
      (if lem-eshell--default-aliases
          (setq eshell-command-aliases-list
                (append lem-eshell--default-aliases
                        lem-eshell-aliases))
        (setq eshell-command-aliases-list lem-eshell-aliases)))))
#+end_src

** Decorating listings

=lem-eshell-better-ls= rewrites each entry Eshell's =ls= prints: it adds a nerd-icons icon, makes the name clickable with the keymap above and a hover highlight, stores the full path, and appends =/= for directories or =*= for executables (skipped for root, who can execute anything) in a dim face.

#+begin_src emacs-lisp -i
(defun lem-eshell-better-ls (file)
  "Add features to listings in `eshell/ls' output.
The features are:
1. Add decoration like 'ls -F':
 ,* Mark directories with a `/'
 ,* Mark executables with a `*'
2. Make each listing into a clickable link to open the
corresponding file or directory.
3. Add icons (requires `nerd-icons`)
This function is meant to be used as advice around
`eshell-ls-annotate', where FILE is the cons describing the file."
  (let* ((name (car file))
         (icon (if (eq (cadr file) t)
                   (nerd-icons-icon-for-dir name)
                 (nerd-icons-icon-for-file name)))
         (suffix
          (cond
           ;; Directory
           ((eq (cadr file) t)
            "/")
           ;; Executable
           ((and (/= (user-uid) 0) ; root can execute anything
                 (eshell-ls-applicable (cdr file) 3 #'file-executable-p (car file)))
            "*"))))
    (cons
     (concat " "
             icon
             " "
             (propertize name
                         'keymap eshell-ls-file-keymap
                         'mouse-face 'highlight
                         'file-name (expand-file-name (substring-no-properties (car file)) default-directory))
             (when (and suffix (not (string-suffix-p suffix name)))
               (propertize suffix 'face 'shadow)))
     (cdr file)
     )))
#+end_src

** Is Eshell visible?

=lem-is-eshell-toggled= checks whether a window shows an Eshell buffer for the current project or directory. It builds names in a different format from =lem-toggle-eshell= (full path, no bullet), so it would not find the buffers that command creates, and nothing calls it.

#+begin_src emacs-lisp -i
(defun lem-is-eshell-toggled ()
  "Checks if eshell is toggled."
  (let ((eshell-buffer-name nil)
        (result nil))
    (if (project-current)
        (setq eshell-buffer-name (concat "*eshell " (project-root (project-current)) "*"))
      (setq eshell-buffer-name (concat "*eshell " default-directory "*")))

      (if (string= eshell-buffer-name (buffer-name (window-buffer elmnt)))
          (setq result t)))
    result))

** Project-aware Eshell toggle

=lem-toggle-eshell= names the Eshell buffer after the current project, or the current directory, as in =*eshell • dotfiles*=.

- If that buffer is visible and you are in Eshell, it closes the window.
- If it is visible but you are elsewhere, it jumps to it.
- Otherwise it opens it, in the home directory with a prefix argument.
- It loads this module on demand, so it can be bound before Eshell is used.
- No keybinding is set here.
- The docstring contains a blank line, so the form spans two units.

#+begin_src emacs-lisp -i
;; Open an eshell in current dir, with project as name. If called with universal
;; arg, open in home dir. With popper this allows for a popup eshell buffer
(defun lem-toggle-eshell (&optional arg)
  "Open `eshell' in current dir, with project as name.
If called with universal arg ARG, open in home dir.

If closed, toggle open and jump to buffer.
If open, and not in eshell, jump to eshell.
If open and in eshell, toggle closed."
  (interactive "P")
  ;; Load eshell configuration if not already loaded
  (unless (featurep 'sp-eshell)
    (require 'sp-eshell))
  (require 'eshell)
  (let ((eshell-exists nil)
        (eshell-buffer-name nil)
        (eshell-buffer nil)
        (default-directory default-directory))

    (if (project-current)
        (setq eshell-buffer-name (concat "*eshell • " (file-name-nondirectory (directory-file-name (project-root (project-current)))) "*"))
      (setq eshell-buffer-name (concat "*eshell • " (file-name-nondirectory (directory-file-name default-directory)) "*")))

    (dolist (buffer (buffer-list) eshell-exists)
      (if (string= (buffer-name buffer) eshell-buffer-name)
          (progn (setq eshell-exists t)
                 (setq eshell-buffer buffer))))

    (cond ((and (get-buffer-window eshell-buffer-name)
                (derived-mode-p 'eshell-mode))
           (pop-to-buffer eshell-buffer-name)
           (delete-window))
          ((and (get-buffer-window eshell-buffer-name)
                (not (derived-mode-p 'eshell-mode)))
           (pop-to-buffer eshell-buffer-name))
          (arg
           (lem-eshell-home))
          (t
           (eshell)))))
#+end_src

(defun eshell/less (&rest args)
  "Invoke `view-file' on a file. \"less +42 foo\" will go to line 42 in

  (while args

	           (file (pop args)))

** z and cg

Directory jumping:

- =z= jumps to a previously visited directory. With no argument and =consult-dir= loaded, it adds Eshell's directory history as a source and picks with =consult-dir=. Otherwise it matches a regexp against history or offers plain completion.
- =cg= jumps to the git root of the current directory.

#+begin_src emacs-lisp -i
;;;;;; Jump Directories (w/Consult & Consult-Dir)
(defun eshell/z (&optional regexp)
  "Navigate to a previously visited directory in eshell, or to
any directory proferred by `consult-dir'."
  (let ((eshell-dirs (delete-dups
                      (mapcar 'abbreviate-file-name
                              (ring-elements eshell-last-dir-ring)))))
    (cond
     ((and (not regexp) (featurep 'consult-dir))
      (let* ((consult-dir--source-eshell `(:name "Eshell"
                                           :narrow ?e
                                           :category file
                                           :face consult-file
                                           :items ,eshell-dirs))
             (consult-dir-sources (cons consult-dir--source-eshell
                                        consult-dir-sources)))
        (eshell/cd (substring-no-properties
                    (consult-dir--pick "Switch directory: ")))))
     (t (eshell/cd (if regexp (eshell-find-previous-directory regexp)
                     (completing-read "cd: " eshell-dirs)))))))

** Viewing files from Eshell

=lem-eshell-view-file= opens a file in View mode and arranges for quitting View mode to kill the buffer and return to Eshell.

- Files whose major mode is special are just shown without View mode.
- This is the function behind =less=.

#+begin_src emacs-lisp -i
(defun lem-eshell-view-file (file)
  "A version of `view-file' which properly respects the eshell prompt."
  (interactive "fView file: ")
  (unless (file-exists-p file) (error "%s does not exist" file))
  (let ((had-a-buf (get-file-buffer file))
	    (buffer (find-file-noselect file)))
    (if (eq (with-current-buffer buffer (get major-mode 'mode-class))
	        'special)
	    (progn
	      (switch-to-buffer buffer)
	      (message "Not using View mode because the major mode is special"))
      (let ((undo-window (list (window-buffer) (window-start)
			                   (+ (window-point)
				                  (length (funcall eshell-prompt-function))))))
	    (switch-to-buffer buffer)
	    (view-mode-enter (cons (selected-window) (cons nil undo-window))
			             'kill-buffer)))))
#+end_src

* Mode hooks

** Eshell buffer setup

=sp-eshell= runs in every Eshell buffer. It binds =C-l= to clear, adds an imenu entry for each prompt so you can jump between past commands, turns off =hl-line-mode=, removes fringes and adds a one-column margin, turns on =visual-line-mode=, and changes the truncation glyph.

Two of these have global side effects: =semantic-mode= is a global mode, so turning it off here disables it everywhere, and =set-display-table-slot= modifies =standard-display-table= for all buffers (and would error if that table is nil).

#+begin_src emacs-lisp -i
;;;;; Setup Hooks
(defun sp-eshell ()
  (interactive)
  ;; Clear eshell keybind
  (local-set-key (kbd "C-l") 'eshell-clear-buffer)
  ;; Use imenu to jump prompts
  ;; https://xenodium.com/imenu-on-emacs-eshell/
  (setq-local imenu-generic-expression
              '(("Prompt" " λ \\(.*\\)" 1)))
  ;; Turn off semantic-mode in eshell buffers
  (semantic-mode -1)
  ;; Turn off hl-line-mode
  (hl-line-mode -1)
  ;; Remove fringe
  (set-window-fringes nil 0 0)
  (set-window-margins nil 1 nil)
  ;; Scrolling
  (setq hscroll-margin 0)
  ;; Text wrapping
  (visual-line-mode +1)
  (set-display-table-slot standard-display-table 0 ?\ ))
(add-hook 'eshell-mode-hook #'sp-eshell)
#+end_src
