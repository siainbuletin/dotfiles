Otherwise all of the links would end up completely broken.

Someone asked for such a function on GitHub, but it never got implemented into the base package. The same person also tried writing a function that would take care of this outside of package.

It doesn't actually work becase the function =denote-org-extras--increase-headings-depth= hasn't actually been replicated again in =denote=.

#+begin_src emacs-lisp :tangle no :eval never
(defvar denote-org-extras-not-ask-merge-delete t
  "If non-nil, `denote-org-extras-merge-note-to-another' deletes
without prompting. Otherwise, prompt for confirmation before
deletion.")

(defun denote-org-extras-merge-note-to-another (file)
  "Merge the current note into another one"
  (interactive (list (denote-file-prompt "\\.org\\'")))
  (if (derived-mode-p 'org-mode)
      (let* ((current-note buffer-file-name)
             (current-note-title (denote-retrieve-front-matter-title-value current-note 'org))
             (current-note-keywords (denote-extract-keywords-from-path current-note))
             (current-note-content (condition-case nil ;; handling case when current note is empty
                                       (denote-org-dblock--get-file-contents current-note t)
                                     (error "")))
             (current-note-as-subtree (format "\n* %s\n%s" current-note-title (denote-org-extras--increase-headings-depth current-note-content)))
             (other-note-keywords (denote-extract-keywords-from-path file))
             (other-note-buffer (find-file-noselect file))
             (other-note-title (denote-retrieve-front-matter-title-value file 'org))
             (all-keywords (delete-dups (append current-note-keywords other-note-keywords))))
        (with-current-buffer other-note-buffer
          (goto-char (point-max))
          (insert current-note-as-subtree))
        (denote-org-extras--delete-note current-note)
        (switch-to-buffer other-note-buffer)
        (denote-rewrite-keywords file all-keywords 'org)
        (denote-rename-file file other-note-title all-keywords nil nil) ;; FIXME: denote-slug-put-equals: Wrong type argument: arrayp, nil
        (message "Merged current note into %s" (file-name-nondirectory file)))
    (user-error "The current note is not an Org file")))
#+end_src

I asked on GitHub for a similar feature for this. Prot wrote a version that aborts the merge between two denote notes if the =OTHER_FILE= has backlinks to it. *Is there really no way to deal with all of the identifiers?*

#+begin_src emacs-lisp :tangle no :eval never
(defun denote-org--merge-get-file-data-without-front-matter (file)
  "Return data of FILE without its front matter.
Data is of the form (TITLE FILETAGS CONTENTS), all of which are strings.

Assume that front matter is a contiguous region of lines starting with
#+ from the beginning of the buffer and ending in an empty line."
  (with-temp-buffer
    (insert-file-contents file)
    (org-mode)
    (goto-char (point-min))
    (let ((title (org-get-title))
          (tags (org-get-tags))
          (contents (progn
                      (when-let* ((beginning (when (re-search-forward "^#\\+" nil t)
                                               (line-beginning-position)))
                                  (end (re-search-forward "^$" nil t)))
                        (delete-region beginning end))
                      (string-trim (buffer-string)))))
      (list title tags contents))))

;;;###autoload
(defun denote-org-merge-notes (target-file other-file)
  "Append OTHER-FILE to TARGET-FILE and delete OTHER-FILE.
When called interactively TARGET-FILE is the current file.
TARGET-FILE and OTHER-FILE are both derived from `org-mode'.

Abort the operation if OTHER-FILE has backlinks."
  (interactive
   (list
    buffer-file-name
    (denote-file-prompt
     "\\.org"
     (format "FILE to merge into `%s'" (propertize buffer-file-name 'face 'denote-faces-prompt-current-name)))))
  (dolist (file (list target-file other-file))
    (with-current-buffer (find-file-noselect file)
      (unless (derived-mode-p 'org-mode)
        (error "The file `%s' is not using `org-mode'" file))))
  (pcase-let ((`(,title ,tags ,contents) (denote-org--merge-get-file-data-without-front-matter other-file)))
    (if (denote--file-has-backlinks-p other-file)
        (error "The other file `%s' has backlinks; aborting the merge" other-file)
      (with-current-buffer (find-file-noselect target-file)
        (goto-char (point-max))
        (unless (looking-at "^$")
          (insert "\n\n"))
        (insert (format "* MERGED: %s :%s:" title (string-join tags ":")))
        (insert "\n\n")
        (insert contents))
      (when (y-or-n-p (format "Merged `%s'; now DELETE it and its buffer? " (propertize other-file 'face 'denote-faces-prompt-current-name)))
        (kill-buffer (get-file-buffer other-file))
        (delete-file other-file)))))
#+end_src
