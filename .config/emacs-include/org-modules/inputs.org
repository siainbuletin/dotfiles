* TODO Configuring RSS Feeds for =elfeed=

Taken directly from Emacs Writing Studio.

#+begin_src emacs-lisp :tangle "init.el" :tangle never
  (use-package elfeed
    :custom
    (elfeed-db-directory
     (expand-file-name "elfeed" user-emacs-directory))
    (elfeed-show-entry-switch 'display-buffer)
    :bind
    ("C-c w e" . elfeed))
#+end_src

* Configuring =elfeed= to be integrated with =org-mode=

Taken directly from Emacs Writing Studio.

#+begin_src emacs-lisp :tangle "init.el" :eval never
  (use-package elfeed-org
    :config
    (elfeed-org)
    :custom
    (rmh-elfeed-org-files
     (list (concat (file-name-as-directory (getenv "HOME"))
  		         "elfeed.org"))))
#+end_src
