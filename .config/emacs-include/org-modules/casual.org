Casual is actually quite annoying and gets in the way a lot, so now it's disabled. I want to be able to see the full image without this big menu always visible.

#+begin_src emacs-lisp :tangle no :eval never
  ;; configure image-mode to have casual bindings
  (use-package image-mode
    :ensure nil
    :after casual
    :config
    (require 'casual-image)

    :hook
    (image-mode . casual-image-tmenu)

    :bind
    (:map image-mode-map
          ("C-o" . casual-image-tmenu)))

  ;; Configure dired to have casual bindings
  (use-package dired
    :ensure nil
    :after casual

    :bind
    (:map dired-mode-map
          ("C-o" . casual-dired-tmenu)))
#+end_src
